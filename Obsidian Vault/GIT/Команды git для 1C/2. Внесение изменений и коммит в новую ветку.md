# Отредактируйте нужные файлы в проекте, затем:

	git status
	git diff
# Добавить изменения в индекс:

	git add .
# Создать коммит:

	git commit -m "Краткое описание изменений"
	
# Отправить ветку в удалённый репозиторий (первый раз):

	git push -u origin имя-ветки

# При последующих push в эту же ветку достаточно:

	git push
